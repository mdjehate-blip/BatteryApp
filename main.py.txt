from kivy.app import App
from kivy.uix.boxlayout import BoxLayout
from kivy.uix.label import Label
from kivy.clock import Clock
from kivy.utils import platform

class BatteryDashboard(BoxLayout):
    def __init__(self, **kwargs):
        super().__init__(orientation='vertical', padding=30, spacing=15, **kwargs)
        
        self.title_label = Label(text="Battery Monitor", font_size='28sp', bold=True, size_hint=(1, 0.2))
        self.status_label = Label(text="Reading battery data...", font_size='20sp', halign="center")
        
        self.add_widget(self.title_label)
        self.add_widget(self.status_label)
        
        # প্রতি ১ সেকেন্ড পর পর ডেটা আপডেট হবে
        Clock.schedule_interval(self.update_battery_info, 1)

    def update_battery_info(self, dt):
        if platform == 'android':
            try:
                from jnius import autoclass, cast
                PythonActivity = autoclass('org.kivy.android.PythonActivity')
                Intent = autoclass('android.content.Intent')
                IntentFilter = autoclass('android.content.IntentFilter')
                BatteryManager = autoclass('android.os.BatteryManager')
                Context = autoclass('android.content.Context')

                activity = PythonActivity.mActivity
                ifilter = IntentFilter(Intent.ACTION_BATTERY_CHANGED)
                battery_status = activity.registerReceiver(None, ifilter)

                # ১. ব্যাটারি পার্সেন্টেজ ও ভোল্টেজ বের করা
                level = battery_status.getIntExtra(BatteryManager.EXTRA_LEVEL, -1)
                scale = battery_status.getIntExtra(BatteryManager.EXTRA_SCALE, -1)
                percentage = int((level / float(scale)) * 100)
                
                voltage = battery_status.getIntExtra(BatteryManager.EXTRA_VOLTAGE, -1) # mV
                temp = battery_status.getIntExtra(BatteryManager.EXTRA_TEMPERATURE, -1) / 10.0 # °C
                
                status = battery_status.getIntExtra(BatteryManager.EXTRA_STATUS, -1)
                is_charging = status == BatteryManager.BATTERY_STATUS_CHARGING or status == BatteryManager.BATTERY_STATUS_FULL
                state_str = "Charging ⚡" if is_charging else "Discharging 🔋"

                # ২. লাইভ কারেন্ট (mA) এবং পাওয়ার (Watt) বের করা
                bm = cast('android.os.BatteryManager', activity.getSystemService(Context.BATTERY_SERVICE))
                current_now = bm.getIntProperty(BatteryManager.BATTERY_PROPERTY_CURRENT_NOW)
                
                # শাওমি বা কিছু ফোনে মাইক্রো-অ্যাম্পিয়ার (uA) রিটার্ন করে, তাই mA-তে রূপান্তর করা
                current_ma = current_now / 1000.0 if abs(current_now) > 10000 else float(current_now)
                power_watt = abs((current_ma / 1000.0) * (voltage / 1000.0))

                self.status_label.text = (
                    f"Status: {state_str}\n\n"
                    f"Battery Level: {percentage}%\n"
                    f"Electric Current: {int(current_ma)} mA\n"
                    f"Power: {power_watt:.1f} W\n"
                    f"Voltage: {voltage} mV\n"
                    f"Temperature: {temp} °C"
                )
            except Exception as e:
                self.status_label.text = f"Error: {str(e)}"
        else:
            # পিসিতে বা সাধারণ পরিবেশে টেস্ট করার জন্য ডেমো লেখা
            self.status_label.text = (
                "Status: Discharging (PC/Test Mode)\n\n"
                "Battery Level: 39%\n"
                "Electric Current: -303 mA\n"
                "Power: 1.2 W\n"
                "Voltage: 3830 mV\n"
                "Temperature: 33.8 °C"
            )

class BatteryApp(App):
    def build(self):
        return BatteryDashboard()

if __name__ == '__main__':
    BatteryApp().run()
